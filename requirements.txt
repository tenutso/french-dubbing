# French Dubbing Pipeline v4.0 — Python Dependencies
# Install: pip install -r requirements.txt
#
# ── PyTorch stack (CUDA 13.x wheels from PyPI; needs NVIDIA driver >= 580) ────
# torch, torchaudio, torchvision and torchcodec are pinned together. torchaudio's
# last release is 2.11.x (audio I/O moved to torchcodec); it only requires
# torch>=2.11 and works with torch 2.15. torchcodec 0.17 is the release paired
# with torch 2.15 — a mismatched torchcodec fails to load its libs on import,
# which surfaces as pyannote "AudioDecoder NameError" + degraded F5-TTS.
# torchvision isn't used directly, but transformers imports it when present and
# base images ship one built for their own torch: a stale build fails with
# "operator torchvision::nms does not exist", which breaks the F5-TTS import.
torch==2.15.0
torchaudio==2.11.1
torchvision==0.29.2
torchcodec==0.17.0

# ── Audio / Video ──────────────────────────────────────────────────────────────
# Upper bound: pyannote.audio 4.x would otherwise pull numpy 2.5.x, which breaks
# numba (requires numpy<2.5) and therefore librosa/demucs.
numpy>=1.26.4,<2.5
scipy>=1.13.1
librosa>=0.10.2
soundfile>=0.12.1
pydub>=0.25.1

# ── Source Separation — Demucs ─────────────────────────────────────────────────
demucs>=4.0.0

# ── Transcription — faster-whisper (CTranslate2) ──────────────────────────────
faster-whisper>=1.1.0
# faster-whisper (<=1.2.1) decodes files with av.open(..., metadata_errors=...),
# an argument PyAV 19 removed → "open() got an unexpected keyword argument
# 'metadata_errors'" at transcription. Cap PyAV until faster-whisper catches up.
av<19

# ── Speaker Diarization — pyannote.audio (optional, gated HF model) ───────────
# 4.x is required by the speaker-diarization-community-1 model and uses the
# token= kwarg + torchcodec audio I/O (see torchcodec pin above).
pyannote.audio>=4.0,<5.0

# ── Speaker Denoising (noisereduce primary, FFmpeg anlmdn fallback) ────────────
# DeepFilterNet was dropped: it has no cp312 wheel (needs a Rust toolchain to
# build) and pins numpy<2.0, conflicting with the numpy 2.x stack. noisereduce
# gives a clean voice-clone reference; FFmpeg anlmdn is the always-present fallback.
noisereduce>=3.0.0

# ── TTS — F5-TTS (flow-matching zero-shot voice cloning, 24 kHz) ─────────────
# MIT; multilingual (incl. French) flow-matching DiT; more reliable than
# autoregressive XTTS-v2 on names and unusual input (no temperature drift).
f5-tts>=1.0.0
transformers>=4.57

# ── Subtitles ─────────────────────────────────────────────────────────────────
pysrt>=1.1.2
# pysrt pulls chardet unpinned (grabs 7.x). requests 2.32.x only recognises
# chardet<6, so a 7.x install triggers a RequestsDependencyWarning on every
# `import requests`. Cap it to keep requests' charset check happy.
chardet<6

# ── Translation transport — Ollama HTTP ──────────────────────────────────────
requests>=2.32.0

# ── HuggingFace download acceleration ─────────────────────────────────────────
# Rust-based transfer client. Activated by HF_HUB_ENABLE_HF_TRANSFER=1.
hf_transfer>=0.1.6

# ── Web UI (FastAPI + uvicorn) ────────────────────────────────────────────────
# Launcher: bash 05_web.sh  → uvicorn web.app:app on 0.0.0.0:7860
fastapi>=0.110
uvicorn[standard]>=0.27
python-multipart>=0.0.9

# ── Syllabification — pyphen (French syllable budgeting) ─────────────────────
pyphen>=0.14.0

# ── Utilities ─────────────────────────────────────────────────────────────────
python-dotenv>=1.0.0
tqdm>=4.66.0
click>=8.1.7
pyyaml>=6.0.1
packaging>=23.0
